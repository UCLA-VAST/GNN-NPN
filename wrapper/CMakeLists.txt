set(MOCKTURTLE_WRAPPER_TARGET mockturtle_wrapper)

pybind11_add_module(${MOCKTURTLE_WRAPPER_TARGET} MODULE mockturtle.cpp)

target_compile_features(${MOCKTURTLE_WRAPPER_TARGET} PRIVATE cxx_std_17)
target_include_directories(${MOCKTURTLE_WRAPPER_TARGET} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(${MOCKTURTLE_WRAPPER_TARGET} PRIVATE mockturtle)

set_target_properties(${MOCKTURTLE_WRAPPER_TARGET} PROPERTIES
  LIBRARY_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/python"
  RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/python"
)

install(TARGETS ${MOCKTURTLE_WRAPPER_TARGET}
  LIBRARY DESTINATION .
  RUNTIME DESTINATION .
)
